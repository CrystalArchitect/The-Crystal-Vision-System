import pathlib
class PersistentStateStore:
    def __init__(self, path=":memory:"):
        self.path=pathlib.Path(path) if str(path)!=":memory:" else None
        self.goals=[]; self.expected={}; self.actual={}; self.history=[]
    def set_goal(self,g): self.goals.append(g)
    def set_expected(self,e): self.expected=e
    def update_actual(self,a): self.actual=a

### FILE: core/gap.py

from dataclasses import dataclass, asdict
@dataclass
class Gap:
    id:int; expected:object; actual:object; magnitude:float; kind:str
    def to_dict(self): return asdict(self)
class GapDetector:
    def __init__(self): self._c=0
    def detect(self, exp, act, kind="mismatch"):
        if exp==act: return None
        self._c+=1; mag=1.0
        if isinstance(exp,dict) and isinstance(act,dict):
            keys=set(exp)|set(act); mag=float(sum(1 for k in keys if exp.get(k)!=act.get(k)))/max(1,len(keys))
            if mag==0: return None
        return Gap(self._c,exp,act,mag,kind)

```

from typing import List, Optional
from .gap import Gap
class AffectModel:
    def classify(self, hist:List[Optional[Gap]]):
        gaps=[g for g in hist if g is not None]
        if not gaps: return "closed"
        if len(hist)>=2 and hist[-1] is not None and hist[-2] is None: return "reopened"
        if len(gaps)>=3 and gaps[-3].expected==gaps[-2].expected==gaps[-1].expected: return "stalled"
        return "uncertain"

from dataclasses import dataclass
@dataclass
class ClosureDecision:
    strategy:str; reason:str; outcome:str="pending"
    def to_dict(self): return {"strategy":self.strategy,"reason":self.reason,"outcome":self.outcome}
class ClosurePolicy:
    def decide(self,label,ctx=None):
        if label=="closed": return ClosureDecision("stop","gap closed","closed")
        if label=="stalled": return ClosureDecision("switch_tool","stalled","switched")
        if label=="reopened": return ClosureDecision("rephrase","gap reopened","rephrased")
        return ClosureDecision("ask","need clarification","asked")

import json, pathlib
class CycleLogger:
    def __init__(self,path):
        self.path=pathlib.Path(path); self.path.parent.mkdir(parents=True, exist_ok=True); self.cycle=0; self.path.write_text("",encoding="utf-8")
    def log(self,op,data):
        self.cycle+=1; e={"cycle":self.cycle,"op":op,"ts":self.cycle,"data":data}
        self.path.open("a",encoding="utf-8").write(json.dumps(e,sort_keys=True)+"\n")

_state=None; _gd=None; _am=None; _cp=None; _log=None
def init_runtime(s,gd,am,cp,lg):
    global _state,_gd,_am,_cp,_log; _state=s; _gd=gd; _am=am; _cp=cp; _log=lg
def track(ev): _state.record(ev); _log.log("track",ev)
def detect_gap(exp,act):
    gap=_gd.detect(exp,act); _log.log("gap_opened" if gap else "gap_none", gap.to_dict() if gap else {"expected":exp,"actual":act}); return gap
def label_affect(hist):
    label=_am.classify(hist); _log.log("label",{"label":label}); return label
def close_with(label,ctx=None):
    d=_cp.decide(label,ctx or {}); _log.log("closure",d.to_dict()); return d

from .state import PersistentStateStore
from .gap import GapDetector
from .affect import AffectModel
from .closure import ClosurePolicy
from .log import CycleLogger
from . import crystalcode
class Loop:
    def __init__(self, store_path=":memory:", log_path="examples/logs/loop.jsonl"):
        self.store=PersistentStateStore(store_path); self.gap_detector=GapDetector(); self.affect_model=AffectModel(); self.closure_policy=ClosurePolicy(); self.logger=CycleLogger(log_path); self.gap_history=[]; crystalcode.init_runtime(self.store,self.gap_detector,self.affect_model,self.closure_policy,self.logger)
    def run_cycle(self, expected, actual, event=None):
        if event: crystalcode.track(event)
        gap=crystalcode.detect_gap(expected,actual); self.gap_history.append(gap)
        label=crystalcode.label_affect(self.gap_history); decision=crystalcode.close_with(label)
        return {"gap":gap,"label":label,"decision":decision}
