### Python: HFT Limit Order Book & Flash Crash Simulation (illustrative, untested here)

```python
import random

class LimitOrderBook:
    def __init__(self, baseline_price=100.0):
        self.bids = {round(baseline_price - i*0.1, 2): 500 for i in range(1, 11)}
        self.asks = {round(baseline_price + i*0.1, 2): 500 for i in range(1, 11)}
        self.mid_price = baseline_price

    def get_highest_bid(self):
        return max(self.bids.keys()) if self.bids else None

    def get_lowest_ask(self):
        return min(self.asks.keys()) if self.asks else None

    def simulate_hft_market_makers(self, network_latency_ms):
        if network_latency_ms > 50:
            print(f"[LATENCY ALERT: {network_latency_ms}ms] HFT Algorithms panicking. Pulling bids...")
            self.bids.clear()
        else:
            highest_bid = self.get_highest_bid() or self.mid_price - 0.1
            lowest_ask = self.get_lowest_ask() or self.mid_price + 0.1
            self.bids[round(highest_bid, 2)] = random.randint(400, 600)
            self.asks[round(lowest_ask, 2)] = random.randint(400, 600)

    def execute_market_sell(self, volume):
        print(f"[MARKET ORDER] Executing urgent Sell Order of {volume} units...")
        while volume > 0:
            highest_bid = self.get_highest_bid()
            if not highest_bid:
                print("[CRASH EFFECT] NO BIDS REMAINING. Price drops to zero! Liquidity Black Hole achieved.")
                self.mid_price = 0.0
                break
            available_vol = self.bids[highest_bid]
            if volume >= available_vol:
                volume -= available_vol
                del self.bids[highest_bid]
                self.mid_price = highest_bid
            else:
                self.bids[highest_bid] -= volume
                volume = 0
                self.mid_price = highest_bid
        print(f"[CURRENT PRICE] Market Mid-Price stabilized at: ${self.mid_price:.2f}\n")

# Run standard LOB simulation execution
lob = LimitOrderBook(baseline_price=100.0)
print("PHASE 1: Normal Operations")
lob.simulate_hft_market_makers(network_latency_ms=2.1)
lob.execute_market_sell(volume=300)
print("PHASE 2: Latency Glitch / Flash Crash")
lob.simulate_hft_market_makers(network_latency_ms=142.5)
lob.execute_market_sell(volume=300)
```

import pandas as pd
import numpy as np
from sqlalchemy import create_engine

def enterprise_data_pipeline(source_type, connection_string, target_table_or_file):
    print(f"Initialising Enterprise Data Pipeline. Connection Type: {source_type.upper()}")
    try:
        if source_type == 'sql':
            engine = create_engine(connection_string)
            query = "SELECT * FROM raw_staging_transactions"
            df = pd.read_sql(query, engine)
        elif source_type == 'excel':
            df = pd.read_excel(connection_string, sheet_name='Q3_Raw_Data')
        else:
            raise ValueError("Invalid source_type. Use 'sql' or 'excel'.")

        print(f"Data ingested successfully. Records found: {df.shape}")
        df = df.drop_duplicates()
        num_cols = df.select_dtypes(include=[np.number]).columns
        cat_cols = df.select_dtypes(include=['object']).columns
        for col in num_cols:
            df[col] = df[col].fillna(df[col].median())
        for col in cat_cols:
            df[col] = df[col].fillna('Unassigned_Record')

        if source_type == 'sql':
            df.to_sql(target_table_or_file, engine, if_exists='replace', index=False)
            print(f"Production table '{target_table_or_file}' successfully written and verified.")
        elif source_type == 'excel':
            df.to_excel(target_table_or_file, index=False, sheet_name='Cleaned_Output')
            print(f"Processed spreadsheet generated successfully at: {target_table_or_file}")
        return True
    except Exception as e:
        print(f"Critical Pipeline Failure: {str(e)}")
        return False
